local OBS_SIG = "HMOBS"
local ZW0, ZW1 = "\u{200B}", "\u{200C}"

local function encodeObs(cmd, uid, extra)
    local s = OBS_SIG .. "|" .. tostring(cmd) .. "|" .. tostring(uid)
    if extra then s = s .. "|" .. tostring(extra):gsub("|", "~") end
    local out = {}
    for i = 1, #s do
        local b = string.byte(s, i)
        for j = 7, 0, -1 do out[#out + 1] = (math.floor(b / 2 ^ j) % 2 == 1) and ZW1 or ZW0 end
    end
    return table.concat(out)
end

local function decodeObs(text)
    if not text or #text < 64 then return nil end
    local bits = {}
    for i = 1, #text do
        local c = string.sub(text, i, i)
        if c == ZW0 then bits[#bits + 1] = "0"
        elseif c == ZW1 then bits[#bits + 1] = "1" end
    end
    if #bits < 8 then return nil end
    local out = {}
    for i = 1, #bits - 7, 8 do
        local b = 0
        for j = 0, 7 do b = b * 2 + tonumber(bits[i + j]) end
        out[#out + 1] = string.char(b)
    end
    local s = table.concat(out)
    if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then return nil end
    local parts = string.split(s, "|")
    if #parts < 3 then return nil end
    local extra = parts[4] and string.gsub(parts[4], "~", "|") or nil
    return parts[2], tonumber(parts[3]), extra
end

local function setupM2Listener(p)
    if p == LocalPlayer then return end
    local function onChar(char)
        if not char then return end
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not hum or not root then return end
        hum.AnimationPlayed:Connect(function(track)
            if not isBlockActive then return end
            local animId = track.Animation and track.Animation.AnimationId
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx, dy, dz = cPos.X-myPos.X, cPos.Y-myPos.Y, cPos.Z-myPos.Z
                    if (dx*dx+dy*dy+dz*dz) <= COMBAT_RANGE_SQ then
                        activeM2Count = activeM2Count + 1
                        rawUnblock()
                        local cleaned = false
                        local sc, ec, dc, tc
                        local function cleanup()
                            if cleaned then return end
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count-1)
                            if sc then sc:Disconnect() end
                            if ec then ec:Disconnect() end
                            if dc then dc:Disconnect() end
                            if tc then task.cancel(tc) end
                        end
                        sc = track.Stopped:Connect(cleanup)
                        ec = track.Ended:Connect(cleanup)
                        dc = hum.Died:Connect(cleanup)
                        tc = task.delay((track.Length>0 and track.Length or 1.5)+0.1, cleanup)
                    end
                end
            end
        end)
    end
    if p.Character then onChar(p.Character) end
    p.CharacterAdded:Connect(onChar)
end

local function FindByName(name) if not name or name == "" then return nil end local low = string.lower(name) for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end end return nil end
local function GetTool() local c = LocalPlayer.Character if not c then return nil end local t = c:FindFirstChildOfClass("Tool") if t then return t end local bp = LocalPlayer:FindFirstChild("Backpack") if bp then for _, x in ipairs(bp:GetChildren()) do if x:IsA("Tool") then x.Parent = c return x end end end return nil end
local function HitAll(target) local c = LocalPlayer.Character if not c then return end for _, t in ipairs(c:GetChildren()) do if t:IsA("Tool") then pcall(function() t.Enabled = true t:Activate() end) end end for _, r in ipairs(Remotes) do pcall(function() r:FireServer("M1") end) pcall(function() r:FireServer() end) end for _, r in ipairs(ManualRemotes) do pcall(function() r:FireServer("M1") end) pcall(function() r:FireServer("Attack") end) end if clientSideHitbox and target then pcall(function() clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) end) end if combatRemote then pcall(function() combatRemote:FireServer("M1") end) pcall(function() combatRemote:FireServer("Attack") end) end end

local function IsMonster(model) if not model or not model:IsA("Model") then return false end local hum = model:FindFirstChildOfClass("Humanoid") if not hum or hum.Health <= 0 then return false end if model == LocalPlayer.Character then return false end if Players:GetPlayerFromCharacter(model) then return false end return (model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart) ~= nil end
local function GetMonsterRoot(model) return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart end
local function ScanMonsters() local list = {} for _, v in ipairs(workspace:GetDescendants()) do if v:IsA("Model") and IsMonster(v) then table.insert(list, v) end end return list end
local function FindTargetByNamePartial(partial) if not partial or partial == "" then return nil end local low = string.lower(partial) for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end end return nil end
local function ClaimOwnership(model) pcall(function() for _, part in ipairs(model:GetDescendants()) do if part:IsA("BasePart") and part.SetNetworkOwner then part:SetNetworkOwner(LocalPlayer) end end end) end

local function CollectAllMonsters()
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        if root then
            ClaimOwnership(model)
            local angle = (count / math.max(1, #found)) * math.pi * 2
            local pos = myRoot.Position + Vector3.new(math.cos(angle) * Config.MonsterOrbitRadius, Config.MonsterOrbitHeight, math.sin(angle) * Config.MonsterOrbitRadius)
            pcall(function() model:PivotTo(CFrame.new(pos)) local hum = model:FindFirstChildOfClass("Humanoid") if hum then hum.WalkSpeed = 60 end end)
            monsters[model] = true
            count = count + 1
        end
    end
end

local function SummonAgainstTarget()
    local target = Config.MonsterTarget
    if not target or not target.Character then return end
    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
    if not tRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        local hum = model:FindFirstChildOfClass("Humanoid")
        if root and hum then
            ClaimOwnership(model)
            if Config.MonsterMode == "TP" then
                local off = Vector3.new(math.random(-3, 3), 2, math.random(-3, 3))
                pcall(function() model:PivotTo(CFrame.new(tRoot.Position + off)) end)
            end
            monsters[model] = true
            count = count + 1
        end
    end
end

    while true do
        task.wait(0.02)
        pcall(function()
            if not State.AutoKillServer then AutoKillStatus.Text = "Trạng thái: OFF ⚪\nVòng: " .. autoKillCycleCount autoKillTargetPlayer = nil return end
            local myChar = LocalPlayer.Character local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart") local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
            if not myRoot or not myHum or myHum.Health <= 0 then return end
            local validPlayers = {}
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local tRoot = p.Character:FindFirstChild("HumanoidRootPart") local tHum = p.Character:FindFirstChildOfClass("Humanoid")
                    if tRoot and tHum and tHum.Health > 0 then local d = (myRoot.Position - tRoot.Position).Magnitude if d <= Config.AutoKillRange then table.insert(validPlayers, {player=p, root=tRoot, hum=tHum, dist=d}) end end
                end
            end
            if #validPlayers == 0 then AutoKillStatus.Text = "ON 🟢 | Không có target\nVòng: " .. autoKillCycleCount autoKillTargetPlayer = nil pcall(function() myRoot.AssemblyLinearVelocity = Vector3.new(0, myRoot.AssemblyLinearVelocity.Y, 0) end) return end
            table.sort(validPlayers, function(a, b) return a.dist < b.dist end)
            local currentTarget = nil
            if autoKillTargetPlayer then for _, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentTarget = v break end end end
            if not currentTarget then currentTarget = validPlayers[1] autoKillTargetPlayer = currentTarget.player end
            if currentTarget and currentTarget.root then
                local head = currentTarget.player.Character:FindFirstChild("Head") local headPos = head and head.Position or currentTarget.root.Position
                local flyPos = headPos + Vector3.new(0, Config.AutoKillHeight, 0) local currentPos = myRoot.Position local dir = flyPos - currentPos local distToTarget = dir.Magnitude
                if distToTarget > 1.5 then
                    local speed = math.min(Config.AutoKillSpeed, distToTarget * 8)
                    pcall(function() myRoot.AssemblyLinearVelocity = dir.Unit * speed myRoot.CFrame = CFrame.new(currentPos) * CFrame.lookAt(currentPos, headPos).Rotation end)
                else
                    pcall(function() myRoot.AssemblyLinearVelocity = Vector3.new(0, 2, 0) myRoot.CFrame = CFrame.lookAt(flyPos, headPos) end)
                    local now = tick()
                    if now >= autoKillNextAttack then
                        autoKillNextAttack = now + Config.AutoKillDelay
                        HitAll(currentTarget.player)
                        if combatRemote then pcall(function() combatRemote:FireServer("M1") combatRemote:FireServer("Attack", currentTarget.player) end) end
                        pcall(function() if currentTarget.hum and currentTarget.hum.Health > 0 then currentTarget.hum:TakeDamage(5) end end)
                        local currentIdx = 1 for i, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentIdx = i break end end
                        local nextIdx = currentIdx + 1 if nextIdx > #validPlayers then nextIdx = 1 autoKillCycleCount = autoKillCycleCount + 1 end
                        autoKillTargetPlayer = validPlayers[nextIdx].player
                    end
                end
                local mode = distToTarget > 1.5 and "✈️ BAY" or "⚔️ ĐÁNH"
                AutoKillStatus.Text = string.format("ON 🟢 | %s\nTarget: %s | Vòng: %d", mode, autoKillTargetPlayer and autoKillTargetPlayer.Name or "-", autoKillCycleCount)
                if (currentPos - autoKillLastPos).Magnitude < 0.5 then autoKillStuckTimer = autoKillStuckTimer + 0.02 if autoKillStuckTimer > 2 then autoKillStuckTimer = 0 local currentIdx = 1 for i, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentIdx = i break end end local nextIdx = currentIdx + 1 if nextIdx > #validPlayers then nextIdx = 1 end autoKillTargetPlayer = validPlayers[nextIdx].player end else autoKillStuckTimer = 0 end
                autoKillLastPos = currentPos
            end
        end)
    end
end)
